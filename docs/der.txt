// Define a estrutura do banco de dados do projeto de verificação de peças

Table usuarios {
  id integer [primary key]
  nome varchar
  email varchar [unique]
  senha_hash varchar
  perfil varchar // OPERADOR, ADMIN
}

Table esteiras {
  id integer [primary key]
  nome_identificador varchar
  velocidade_atual integer
  status varchar // ATIVA, PAUSADA, MANUTENCAO
  data_alteracao timestamp
}

Table dados_producao {
  id integer [primary key]
  esteira_id integer [not null]
  data_hora_inicio timestamp
  total_processadas integer
  total_aprovadas integer
  total_recusadas integer
}

Table pecas_recusadas {
  id integer [primary key]
  esteira_id integer [not null]
  operador_id integer
  data_hora timestamp
  imagem_url varchar [note: 'Caminho/URL da foto salva no servidor Docker']
  erro_identificado varchar
  nivel_acuracia decimal
}

// Relacionamentos

Ref: esteiras.id < dados_producao.esteira_id // 1:N

Ref: esteiras.id < pecas_recusadas.esteira_id // 1:N

Ref: usuarios.id < pecas_recusadas.operador_id // 1:N
